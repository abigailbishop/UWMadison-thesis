import ROOT
import numpy as np
import matplotlib.pyplot as plt
run = 16766
file = ROOT.TFile.Open(f"/data/ana/ARA/processing/data/10pct/L2/station_2/year_2020/station_2_run_000{run}.root")
plot_title = f"ARA2 Run {run}"
file.results_tree.Draw("unixtime:avg_snr_v")
n_events = file.results_tree.GetEntries()
x_array = np.copy(np.frombuffer( file.results_tree.GetV1(), np.dtype(float), n_events))
x_array = (x_array - x_array[0])/60
y_array = np.copy(np.frombuffer( file.results_tree.GetV2(), np.dtype(float), n_events))
fig, ax = plt.subplots()
H, xbins, ybins = np.histogram2d(x_array, y_array, bins=180)
extent = [xbins[0], xbins[-1], ybins[0], ybins[-1]]
mappable = ax.imshow(H.T, extent=extent, cmap='viridis', 
                        origin='lower', aspect='auto')
ax.set_xlabel("Time Since Start of Run [min]")
ax.set_ylabel("Average SNR in VPol Antennas")
ax.set_title(plot_title)
plt.colorbar(mappable=mappable, label="Number of Events")
plt.tight_layout()
plt.savefig(f"a2_r{run}_snrVStime.pdf")
plt.savefig(f"a2_r{run}_snrVStime.png", dpi=400)
